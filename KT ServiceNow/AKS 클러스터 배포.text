### aks 배포 전, encryption-at-host 적용을 위한 설정 (적용 안되어있으면 아래 명령어 수행)
az feature show --namespace Microsoft.Compute --name EncryptionAtHost --query properties.state   ## 등록 상태 확인 (registered 확인 필요)
az feature register --name EncryptionAtHost  --namespace Microsoft.Compute                      ## EncryptionAtHost 기능 등록 (몇 분 소요됨)
az provider register --namespace Microsoft.ContainerService                                     ## 리소스 공급자 등록

az aks create `
 --name $CLUSTER_NAME `
 --nodepool-name $NODEPOOL_NAME `
 --resource-group $RESOURCE_GROUP_NAME `
 --tier $CLUSTER_TIER `
 --location $LOCATION `
 --enable-private-cluster `
 --kubernetes-version $KUBE_VERSION `
 --max-pods $MAX_PODS `2
 --node-count $NODE_COUNT `
 --network-plugin azure `
 --network-plugin-mode overlay `
 --vnet-subnet-id /subscriptions/$SUBSCRIPTION_ID/resourceGroups/$NETWORK_RESOURCE_GROUP_NAME/providers/Microsoft.Network/virtualNetworks/$VNET_NAME/subnets/$VNET_SUBNET_APP `
 --pod-cidr $AKS_POD_CIDR `
 --dns-service-ip $AKS_SVC_DNS_IP `
 --service-cidr $AKS_SVC_CIDR `
 --node-vm-size $NODE_SKU `
 --os-sku $NODE_OS `
 --node-osdisk-size $NODE_DISK_SIZE `
 --outbound-type userDefinedRouting `
 --assign-kubelet-identity $FULL_KUBELET_ID `
 --assign-identity $FULL_CONTROL_ID `
 --tags "name=$CLUSTER_NAME" `
 --node-os-upgrade-channel None `
 --private-dns-zone none `
 --enable-cluster-autoscaler `
 --min-count $MIN_NODE_COUNT `
 --max-count $MAX_NODE_COUNT `
 --enable-addons azure-policy `
 --enable-aad  `
 --aad-tenant-id "e6c9ec09-8430-4a99-bf15-242bc089b409"  `
 --enable-azure-rbac  `
 --enable-encryption-at-host `
 --zones 1 2 3  `
 --generate-ssh-keys
